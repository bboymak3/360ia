export interface Env { DB: D1Database; } export default { async fetch(request: Request, env: Env): Promise { // 1. Configurar Cabeceras CORS (Vital para que el Dashboard no de error) const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; // Responder a la validación del navegador (Preflight) if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } if (request.method !== "POST") { return new Response("Método no permitido", { status: 405, headers: corsHeaders }); } try { const body = await request.json() as any; const { widgetId, nuevoContenido, fuente } = body; // Validación básica if (!widgetId || !nuevoContenido) { return new Response(JSON.stringify({ error: "Faltan datos (widgetId o contenido)" }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // 2. Obtener el contexto actual de la DB // Usamos comillas dobles en el nombre de la tabla por si tiene números o caracteres especiales const registro = await env.DB.prepare( "SELECT contexto_entrenamiento FROM 360ia_db WHERE widget_id = ?" ).bind(widgetId).first(); if (!registro) { return new Response(JSON.stringify({ error: "Widget ID no encontrado en la base de datos" }), { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } // 3. Preparar el nuevo bloque de texto const fecha = new Date().toISOString().split('T')[0]; const bloqueNuevo = `\n\n=== INFORMACIÓN ADICIONAL (${fuente} - ${fecha}) ===\n${nuevoContenido}\n`; const contextoFinal = (registro.contexto_entrenamiento || "") + bloqueNuevo; // 4. Actualizar la base de datos await env.DB.prepare( "UPDATE 360ia_db SET contexto_entrenamiento = ? WHERE widget_id = ?" ).bind(contextoFinal, widgetId).run(); return new Response(JSON.stringify({ success: true, message: "Cerebro alimentado" }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } catch (err: any) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } }); } } };
